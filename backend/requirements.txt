# Backend dependencies.

fastapi>=0.115
uvicorn[standard]>=0.32

# pydantic[email] pulls email-validator, which EmailStr needs.
pydantic[email]>=2.9
python-dotenv>=1.0

# Supabase Python client. We use create_async_client from this package, NOT the
# abandoned third-party supabase-py-async on PyPI. Do not install that one.
supabase>=2.31

# Token minting and room management. This is the server SDK, a different
# package from the `livekit` realtime SDK the agent uses.
livekit-api>=1.2

# No Plivo SDK: LiveKit dials through the Plivo SIP trunk, so the backend
# only ever talks to LiveKit (livekit-api above).

# Meeting links (Zoom, Google) and WhatsApp (Meta) are plain HTTPS calls.
httpx>=0.27

# Verifies dashboard sign in tokens against Supabase's public signing key
# (app/auth.py). The crypto extra brings the ES256 support Supabase uses.
PyJWT[crypto]>=2.9

# Timezone database. Windows has none built in, and Python's zoneinfo needs
# it to understand Asia/Kolkata. Harmless on Linux.
tzdata>=2024.1
